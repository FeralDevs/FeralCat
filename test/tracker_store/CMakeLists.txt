cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(meow_tracker_tests LANGUAGES CXX)
  enable_testing()
endif()
set(TRACKER_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_executable(tracker_store_test tracker_store_test.cpp)
add_executable(tracker_monitor_test tracker_monitor_test.cpp
  "${TRACKER_ROOT}/src/app/app_17/tracker_monitor.cpp")
foreach(target tracker_store_test tracker_monitor_test)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "${TRACKER_ROOT}/src/app/app_17")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
target_include_directories(tracker_monitor_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/stubs")
add_test(NAME tracker_store COMMAND tracker_store_test)
add_test(NAME tracker_monitor COMMAND tracker_monitor_test)
set_tests_properties(tracker_store tracker_monitor PROPERTIES TIMEOUT 15)
