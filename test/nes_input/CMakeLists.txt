cmake_minimum_required(VERSION 3.20)
project(nes_input_tests CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
add_executable(nes_input_test nes_input_test.cpp)
if(MSVC)
  target_compile_options(nes_input_test PRIVATE /W4 /WX)
else()
  target_compile_options(nes_input_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME nes_input COMMAND nes_input_test)
