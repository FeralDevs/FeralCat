cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(meow_player_ui_tests LANGUAGES C CXX)
  enable_testing()
endif()
set(UI_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
file(GLOB_RECURSE LVGL_HOST_SOURCES CONFIGURE_DEPENDS "${UI_ROOT}/lib/lvgl/src/*.c")
add_library(lvgl_player_host STATIC ${LVGL_HOST_SOURCES})
set_target_properties(lvgl_player_host PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_compile_definitions(lvgl_player_host PUBLIC LV_CONF_INCLUDE_SIMPLE)
target_include_directories(lvgl_player_host PUBLIC "${CMAKE_CURRENT_LIST_DIR}" "${UI_ROOT}/lib/lvgl")
if(MSVC)
  target_compile_definitions(lvgl_player_host PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(lvgl_player_host PRIVATE /utf-8)
endif()
add_executable(player_ui_test player_ui_test.cpp "${UI_ROOT}/src/app/app_lua/player_ui.cpp"
  "${UI_ROOT}/src/app/app_lua/player_font_12.c"
  "${UI_ROOT}/src/app/app_lua/player_font_18.c"
  "${UI_ROOT}/src/app/app_lua/player_font_20.c")
target_compile_features(player_ui_test PRIVATE cxx_std_17)
set_target_properties(player_ui_test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_include_directories(player_ui_test PRIVATE "${UI_ROOT}/src/app/app_lua")
target_link_libraries(player_ui_test PRIVATE lvgl_player_host)
if(MSVC)
  target_compile_options(player_ui_test PRIVATE /utf-8)
endif()
add_test(NAME player_ui COMMAND player_ui_test)
set_tests_properties(player_ui PROPERTIES TIMEOUT 30)
