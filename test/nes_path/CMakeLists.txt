cmake_minimum_required(VERSION 3.20)
project(nes_path_tests CXX)
enable_testing()
add_executable(nes_path_test nes_path_test.cpp ../../src/system/emulation/nes_path.cpp)
target_compile_features(nes_path_test PRIVATE cxx_std_17)
target_include_directories(nes_path_test PRIVATE ../../src/system/emulation)
if(MSVC)
  target_compile_options(nes_path_test PRIVATE /W4 /WX)
else()
  target_compile_options(nes_path_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME nes_path_boundary COMMAND nes_path_test)
