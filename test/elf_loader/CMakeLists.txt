cmake_minimum_required(VERSION 3.20)
project(elf_loader_lifecycle LANGUAGES C)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(elf_loader_host OBJECT ../../lib/elf_loader/src/esp_elf.c)
target_include_directories(elf_loader_host PRIVATE stubs ../../lib/elf_loader/include)
if(MSVC)
  # Upstream ELF32 casts intentionally narrow host pointers. This suite never
  # executes relocated code; it exercises the real allocation/cleanup paths.
  target_compile_options(elf_loader_host PRIVATE /W0 /experimental:c11atomics)
  target_compile_definitions(elf_loader_host PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(elf_loader_host PRIVATE -w)
  target_compile_definitions(elf_loader_host PRIVATE _GNU_SOURCE)
endif()
add_executable(elf_loader_test loader_lifecycle_test.c $<TARGET_OBJECTS:elf_loader_host>)
target_include_directories(elf_loader_test PRIVATE stubs ../../lib/elf_loader/include)
if(MSVC)
  target_compile_options(elf_loader_test PRIVATE /W4 /WX)
else()
  target_compile_options(elf_loader_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME elf_loader_cleanup COMMAND elf_loader_test)
set_tests_properties(elf_loader_cleanup PROPERTIES TIMEOUT 10)
