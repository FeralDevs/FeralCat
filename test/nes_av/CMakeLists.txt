cmake_minimum_required(VERSION 3.16)
project(nes_av_tests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(nes_av_test nes_av_test.cpp ../../src/system/emulation/nes_av.cpp)
target_compile_features(nes_av_test PRIVATE cxx_std_17)
target_compile_definitions(nes_av_test PRIVATE MEOW_NES_AV_HOST_TEST=1)
target_include_directories(nes_av_test PRIVATE . ../../src/system/emulation)
target_link_libraries(nes_av_test PRIVATE Threads::Threads)
if(WIN32)
    target_link_libraries(nes_av_test PRIVATE winmm)
endif()
if(MSVC)
    target_compile_options(nes_av_test PRIVATE /W4)
else()
    target_compile_options(nes_av_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME nes_av_lifecycle COMMAND nes_av_test)
set_tests_properties(nes_av_lifecycle PROPERTIES TIMEOUT 20)
