cmake_minimum_required(VERSION 3.20)
project(meow_nes_host C)
option(MEOW_NES_ASAN "Enable AddressSanitizer on the host" OFF)
if(MEOW_NES_ASAN)
 if(MSVC)
  add_compile_options(/fsanitize=address /Zi)
 else()
  add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address)
 endif()
endif()
enable_testing()
add_subdirectory(../../lib/nes_core core)
add_executable(nes_core_tests test_core.c)
target_link_libraries(nes_core_tests PRIVATE meow_nes)
add_test(NAME nes_core_contract COMMAND nes_core_tests)
add_executable(nes_core_host host.c)
target_link_libraries(nes_core_host PRIVATE meow_nes)
if(MSVC)
 target_compile_definitions(nes_core_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_definitions(nes_core_host PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
