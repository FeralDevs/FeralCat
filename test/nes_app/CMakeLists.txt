cmake_minimum_required(VERSION 3.20)
project(nes_app_contract LANGUAGES C CXX)
enable_testing()
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(nes_app_test nes_app_test.cpp "../../sd files/apps/nes/app_main.c")
target_include_directories(nes_app_test PRIVATE ../../src/system)
if(MSVC)
  target_compile_options(nes_app_test PRIVATE /W4 /WX)
  target_compile_definitions(nes_app_test PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(nes_app_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME nes_app_contract COMMAND nes_app_test)
set_tests_properties(nes_app_contract PROPERTIES TIMEOUT 10)
