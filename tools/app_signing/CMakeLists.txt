cmake_minimum_required(VERSION 3.16)
project(meowsign LANGUAGES C)
set(MONOCYPHER "${CMAKE_CURRENT_SOURCE_DIR}/../../lib/monocypher/src")
add_executable(meowsign meowsign.c "${MONOCYPHER}/monocypher.c" "${MONOCYPHER}/monocypher-ed25519.c")
target_include_directories(meowsign PRIVATE "${MONOCYPHER}")
set_target_properties(meowsign PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
if(WIN32)
  target_link_libraries(meowsign PRIVATE bcrypt advapi32)
endif()
if(MSVC)
  target_compile_definitions(meowsign PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(meowsign PRIVATE /W4)
else()
  target_compile_options(meowsign PRIVATE -Wall -Wextra)
endif()
