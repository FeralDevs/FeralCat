cmake_minimum_required(VERSION 3.20)
project(meow_nes C)
set(CMAKE_C_STANDARD 11)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS "vendor/nes/*.c" "vendor/mappers/*.c")
add_library(meow_nes STATIC meow_nes.c ${CORE_SOURCES})
target_include_directories(meow_nes PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE vendor)
if(MSVC)
  target_compile_options(meow_nes PRIVATE /O2 /W3)
  target_compile_definitions(meow_nes PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(meow_nes PRIVATE -O2 -Wall -Wno-unused-function -Wno-unused-variable)
  target_link_libraries(meow_nes PUBLIC m)
endif()
